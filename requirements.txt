# On Linux, plain PyPI torch is a multi-GB CUDA build that hard-crashes on GPUs
# it has no kernels for (torch.AcceleratorError on first encode — found on the
# 2026-07-14 clean-venv test). This tool's embeddings are CPU-sized by design,
# so Linux installs pin the CPU wheel from the PyTorch index. Mac/Windows PyPI
# wheels are already CPU-only.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.12.1+cpu; sys_platform == 'linux'
torch==2.12.1; sys_platform != 'linux'

litellm==1.84.0
sentence-transformers==5.5.1

# PDF reading. Was PyPDF2==3.0.1 until 2026-08-31; PyPDF2 is unmaintained under
# that name (a malformed PDF can pin it at full CPU — GHSA-hm9v-vj3r-r55m) and
# pypdf is the maintained continuation with the same PdfReader API. The swap
# CHANGES extracted text: near-identical on almost every source, but pypdf reads
# the letter-spaced class (data/*/sources/anthropic2024.pdf) as clean prose where
# PyPDF2 returned "e p u b l i c l y", so source_decomposer's pdftotext fallback
# fires less. CACHE_SCHEMA 8->9 rebuilds sentence indexes from the new read.
pypdf==6.16.2
